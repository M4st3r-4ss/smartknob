# Dependencies for `py -m smartknob_deck` on Windows.
#
# Install with:
#     py -m pip install -r requirements.txt
#
# The agent speaks the firmware's plaintext host-agent lines, so it needs only
# pyserial plus the Windows backends: pycaw/comtypes for volume, and
# screen-brightness-control (with WMI for laptop panels) for brightness.
#
# The other tools in this repo (smartknob_io.py, simple_example.py) use the
# protobuf protocol instead; install the Pipfile for those.

pyserial==3.5

pycaw==20251023
comtypes==1.4.16
screen-brightness-control==0.27.2
WMI==1.5.1
